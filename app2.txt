    var step = Math.pow(10, Math.floor(Math.log10(Math.max(hi - lo, 1e-9))));
    lo = Math.floor(lo / step) * step; hi = Math.ceil(hi / step) * step;
    return hi === lo ? [lo - step, hi + step] : [lo, hi];
  }

      };

        if (ticking) return; ticking = true;

    });
  }
